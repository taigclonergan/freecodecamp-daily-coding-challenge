'''
Credit Card Masker
Given a string of credit card numbers, return a masked version of it using the following constraints:

The string will contain four sets of four digits (0-9), with all sets being separated by a single space, or a single hyphen (-).
Replace all numbers, except the last four, with an asterisk (*).
Leave the remaining characters unchanged.
For example, given "4012-8888-8888-1881" return "****-****-****-1881".

Tests
1. mask("4012-8888-8888-1881") should return "****-****-****-1881".
2. mask("5105 1051 0510 5100") should return "**** **** **** 5100".
3. mask("6011 1111 1111 1117") should return "**** **** **** 1117".
4. mask("2223-0000-4845-0010") should return "****-****-****-0010".
'''

import re

def mask(card):
    credit_card_pattern = r'^\d{4}([ -])\d{4}\1\d{4}\1\d{4}$' # enforces same separator throughout credit card string
    if not bool(re.match(credit_card_pattern, card)):
        raise ValueError("Not a legitimate credit card input")

    masked_char_list = []
    for char_index in range(19):
        if char_index < 15 and char_index not in (4,9,14):
            masked_char_list.append('*') # non-separator and not in last four digits
        else:
            masked_char_list.append(card[char_index])
    
    masked_card = ''.join(masked_char_list)

    return masked_card
