'''
Landing Spot
In day four of Space Week, you are given a matrix of numbers (an array of arrays), representing potential landing spots for your rover. Find the safest landing spot based on the following rules:

Each spot in the matrix will contain a number from 0-9, inclusive.
Any 0 represents a potential landing spot.
Any number other than 0 is too dangerous to land. The higher the number, the more dangerous.
The safest spot is defined as the 0 cell whose surrounding cells (up to 4 neighbors, ignore diagonals) have the lowest total danger.
Ignore out-of-bounds neighbors (corners and edges just have fewer neighbors).
Return the indices of the safest landing spot. There will always only be one safest spot.
For instance, given:
[
  [1, 0],
  [2, 0]
]
Return [0, 1], the indices for the 0 in the first array.

Tests
1. find_landing_spot([[1, 0], [2, 0]]) should return [0, 1].
2. find_landing_spot([[9, 0, 3], [7, 0, 4], [8, 0, 5]]) should return [1, 1].
3. find_landing_spot([[1, 2, 1], [0, 0, 2], [3, 0, 0]]) should return [2, 2].
4. find_landing_spot([[9, 6, 0, 8], [7, 1, 1, 0], [3, 0, 3, 9], [8, 6, 0, 9]]) should return [2, 1].
'''

def find_landing_spot(matrix):
    best_landing_spot_neighbors = [10,10,10,10] # neighbors in left, right, top, bottom
    best_landing_spot = []

    neighbors = []
    last_row = len(matrix) - 1  # index for last row
    last_col = len(matrix[0]) - 1  # index for last column (zero chosen arbitrarily)
    for row in range(0,last_row + 1):
        for col in range(0,last_col + 1):
            
            if matrix[row][col] != 0:
                continue

            # top left
            if row == 0 and col == 0:
                neighbors = [0, matrix[row][col+1], 0, matrix[row+1][col]]
            # top right
            elif row == 0 and col == last_col:
                neighbors = [matrix[row][col-1], 0, 0, matrix[row+1][col]]
            # bottom left
            elif row == last_row and col == 0:
                neighbors = [0, matrix[row][col+1], matrix[row-1][col], 0]
            # bottom right
            elif row == last_row and col == last_col:
                neighbors = [matrix[row][col-1], 0, matrix[row-1][col], 0]
            # mid top
            elif row == 0:
                neighbors = [matrix[row][col-1], matrix[row][col+1], 0, matrix[row+1][col]]
            # mid left
            elif col == 0:
                neighbors = [0, matrix[row][col+1], matrix[row-1][col], matrix[row+1][col]]
            # mid right
            elif col == last_col:
                neighbors = [matrix[row][col-1], 0, matrix[row-1][col], matrix[row+1][col]]
            # mid bottom
            elif row == last_row:
                neighbors = [matrix[row][col-1], matrix[row][col+1], matrix[row-1][col], 0]
            # central
            else:
                neighbors = [matrix[row][col-1], matrix[row][col+1], matrix[row-1][col], matrix[row+1][col]]


            if sum(neighbors) < sum(best_landing_spot_neighbors): # check for less danger
                best_landing_spot = [row,col]
                best_landing_spot_neighbors = neighbors
            else:
                continue
    return best_landing_spot

def find_landing_spot2(matrix):
    best_landing_spot = None
    best_landing_spot_neighbors = [10, 10, 10, 10]  # arbitrary high danger

    rows = len(matrix)
    cols = len(matrix[0])

    for row in range(rows):
        for col in range(cols):
            if matrix[row][col] != 0:
                continue

            # (left, right, up, down)
            directions = [(0, -1), (0, 1), (-1, 0), (1, 0)]
            neighbors = []

            for dr, dc in directions:
                r, c = row + dr, col + dc
                if 0 <= r < rows and 0 <= c < cols:
                    neighbors.append(matrix[r][c])
                else:
                    neighbors.append(0)

            if sum(neighbors) < sum(best_landing_spot_neighbors):
                best_landing_spot = [row, col]
                best_landing_spot_neighbors = neighbors

    return best_landing_spot


def printing_func2(list_of_matrices):

        print("Matrix of interest")

        print("Safest landing spot")
